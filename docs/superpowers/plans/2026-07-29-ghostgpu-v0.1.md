# ghostgpu v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship ghostgpu v0.1 — an operator that turns kwok fake nodes into simulated GPU nodes by publishing DRA ResourceSlices and legacy extended-resource capacity, so a real kube-scheduler makes real placement decisions against hardware that does not exist.

**Architecture:** A single controller-runtime operator reconciles two cluster-scoped CRDs (`GPUModel`, `GPUPool`). For every kwok-managed node matching a pool's selector, it publishes a DRA `ResourceSlice` (one device per simulated GPU, carrying product/UUID/NVLink-domain attributes) and optionally patches the node's `status.capacity`/`allocatable` with `nvidia.com/gpu` plus GPU Feature Discovery-compatible labels. Pure logic (device identity, slice construction, label construction, safety checks) lives in small, separately-testable packages; the controller only orchestrates them.

**Tech Stack:** Go 1.26.5, kubebuilder v4.15.0, controller-runtime, `k8s.io/api/resource/v1` (DRA, GA), envtest 1.36.2, kind v0.32.0, kwokctl v0.8.0, Docker.

## Global Constraints

- **Development happens in WSL2 Ubuntu 22.04, not native Windows.** kubebuilder ships no Windows binary and envtest requires a Linux `kube-apiserver`. Working copy: `~/repos/ghostgpu`. Toolchain on `PATH` via `~/.local/go/bin`, `~/.local/bin`, `~/go/bin`.
- **Go:** 1.26.5. **kubebuilder:** v4.15.0. **k8s.io/api:** v0.36.3.
- **API group:** `ghostgpu.dev/v1alpha1`. Revisit at the v0.1 public-release gate (see spec §11); it is a find-and-replace while unreleased.
- **DRA driver name:** `gpu.ghostgpu.dev`.
- **envtest Kubernetes version:** 1.36.2. **kind node image:** `kindest/node:v1.36.1`.
- **envtest runs no kube-scheduler.** Controller tests assert objects the operator *writes*. Any assertion about scheduling *decisions* belongs in the e2e suite against kind+kwok.
- **License:** Apache-2.0. All commits use DCO sign-off (`git commit -s`) — required for any future CNCF donation and painful to retrofit.
- **Safety invariant (spec §5):** the operator must never modify a Node lacking the `kwok.x-k8s.io/node` annotation. This is a hard rule, tested.
- **DRA API constraints discovered in the spike:** `sharedCounters` and `devices` cannot coexist in one ResourceSlice; max 128 devices per slice (64 if devices use counters or taints). v0.1 does not use counters, but slice-sharding must not be designed out.

// Package gpu builds the simulated-GPU objects ghostgpu publishes.
package gpu

import (
	"crypto/sha1"
	"encoding/hex"
	"fmt"
)

// DeviceName is the DRA device name for a simulated GPU on a node.
// DRA device names must be valid DNS labels and are unique within a pool,
// and ghostgpu uses one pool per node, so the index alone suffices.
func DeviceName(_ string, index int32) string {
	return fmt.Sprintf("gpu-%d", index)
}

// DeviceUUID derives a stable, NVIDIA-shaped GPU UUID from node and index.
// Deterministic so a restarted operator republishes identical identities
// instead of churning ResourceSlices.
func DeviceUUID(nodeName string, index int32) string {
	sum := sha1.Sum([]byte(fmt.Sprintf("ghostgpu/%s/%d", nodeName, index)))
	h := hex.EncodeToString(sum[:])
	return fmt.Sprintf("GPU-%s-%s-%s-%s-%s", h[0:8], h[8:12], h[12:16], h[16:20], h[20:32])
}

// NVLinkDomain returns the NVLink domain a GPU index belongs to.
// Returns "" when domains are disabled.
func NVLinkDomain(index, domainSize int32) string {
	if domainSize <= 0 {
		return ""
	}
	return fmt.Sprintf("domain-%d", index/domainSize)
}
```

// Package safety enforces that ghostgpu only ever touches simulated infrastructure.
package safety

import corev1 "k8s.io/api/core/v1"

// KwokNodeAnnotation marks a Node as managed by kwok. Its presence is the
// sole signal that a node is safe for ghostgpu to modify.
const KwokNodeAnnotation = "kwok.x-k8s.io/node"

// IsSimulatedNode reports whether a node is kwok-managed and therefore safe
// to modify. ghostgpu must never write to a node where this returns false:
// doing so would mean mutating real infrastructure.
func IsSimulatedNode(node *corev1.Node) bool {
	if node == nil {
		return false
	}
	_, ok := node.Annotations[KwokNodeAnnotation]
	return ok
}
```

package gpu

import (
	"fmt"

	resourcev1 "k8s.io/api/resource/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"

	"github.com/santimillang/ghostgpu/api/v1alpha1"
)

// DriverName is the DRA driver ghostgpu publishes under.
const DriverName = "gpu.ghostgpu.dev"

// SliceName is the ResourceSlice object name for a pool on a node.
func SliceName(poolName, nodeName string) string {
	return fmt.Sprintf("%s-%s", poolName, nodeName)
}

func strAttr(s string) resourcev1.DeviceAttribute {
	v := s
	return resourcev1.DeviceAttribute{StringValue: &v}
}

// BuildResourceSlice constructs the ResourceSlice advertising a pool's
// simulated GPUs on one node. One slice per node, so the pool name in DRA
// terms is the node name — this keeps each slice well under the 128-device
// cap and lets nodes be reconciled independently.
func BuildResourceSlice(pool *v1alpha1.GPUPool, model *v1alpha1.GPUModel, nodeName string) *resourcev1.ResourceSlice {
	devices := make([]resourcev1.Device, 0, pool.Spec.GPUsPerNode)

	for i := int32(0); i < pool.Spec.GPUsPerNode; i++ {
		attrs := map[resourcev1.QualifiedName]resourcev1.DeviceAttribute{
			"productName": strAttr(model.Spec.ProductName),
			"uuid":        strAttr(DeviceUUID(nodeName, i)),
			"vendor":      strAttr(model.Spec.Vendor),
		}
		if domain := NVLinkDomain(i, pool.Spec.Topology.NVLinkDomainSize); domain != "" {
			attrs["nvlinkDomain"] = strAttr(domain)
		}

		devices = append(devices, resourcev1.Device{
			Name:       DeviceName(nodeName, i),
			Attributes: attrs,
			Capacity: map[resourcev1.QualifiedName]resourcev1.DeviceCapacity{
				"memory": {Value: model.Spec.Memory},
			},
		})
	}

	node := nodeName
	return &resourcev1.ResourceSlice{
		ObjectMeta: metav1.ObjectMeta{
			Name: SliceName(pool.Name, nodeName),
		},
		Spec: resourcev1.ResourceSliceSpec{
			Driver:   DriverName,
			NodeName: &node,
			Pool: resourcev1.ResourcePool{
				Name:               nodeName,
				Generation:         1,
				ResourceSliceCount: 1,
			},
			Devices: devices,
		},
	}
}

import (

// NodeLabels returns GPU Feature Discovery-compatible labels for a simulated
// node. Key names and value formats mirror NVIDIA GFD exactly: real tooling
// selects on these, so divergence would silently break compatibility.
func NodeLabels(pool *v1alpha1.GPUPool, model *v1alpha1.GPUModel) map[string]string {
	labels := map[string]string{
		"nvidia.com/gpu.present": "true",
		"nvidia.com/gpu.count":   strconv.Itoa(int(pool.Spec.GPUsPerNode)),
		"nvidia.com/gpu.product": model.Spec.ProductName,

	if major, minor, ok := strings.Cut(model.Spec.ComputeCapability, "."); ok {
		labels["nvidia.com/gpu.compute.major"] = major
		labels["nvidia.com/gpu.compute.minor"] = minor
	}

	return labels
}

// GPUPoolReconciler reconciles a GPUPool object.
type GPUPoolReconciler struct {
	client.Client
	Scheme *runtime.Scheme
}

// +kubebuilder:rbac:groups=ghostgpu.dev,resources=gpupools,verbs=get;list;watch;create;update;patch;delete
// +kubebuilder:rbac:groups=ghostgpu.dev,resources=gpupools/status,verbs=get;update;patch
// +kubebuilder:rbac:groups=ghostgpu.dev,resources=gpumodels,verbs=get;list;watch
// +kubebuilder:rbac:groups=resource.k8s.io,resources=resourceslices,verbs=get;list;watch;create;update;patch;delete
// +kubebuilder:rbac:groups="",resources=nodes,verbs=get;list;watch;update;patch
// +kubebuilder:rbac:groups="",resources=nodes/status,verbs=get;update;patch

func (r *GPUPoolReconciler) Reconcile(ctx context.Context, req ctrl.Request) (ctrl.Result, error) {
	logger := log.FromContext(ctx)

	var pool v1alpha1.GPUPool
	if err := r.Get(ctx, req.NamespacedName, &pool); err != nil {
		return ctrl.Result{}, client.IgnoreNotFound(err)
	}

	var model v1alpha1.GPUModel
	if err := r.Get(ctx, client.ObjectKey{Name: pool.Spec.ModelRef}, &model); err != nil {
		if apierrors.IsNotFound(err) {
			r.setReady(&pool, metav1.ConditionFalse, "ModelNotFound",
				fmt.Sprintf("GPUModel %q not found", pool.Spec.ModelRef))
			return ctrl.Result{}, r.Status().Update(ctx, &pool)
		}
		return ctrl.Result{}, err
	}

	var nodes corev1.NodeList
	if err := r.List(ctx, &nodes, client.MatchingLabels(pool.Spec.NodeSelector)); err != nil {
		return ctrl.Result{}, err
	}

	var matched, devices int32
	for i := range nodes.Items {
		node := &nodes.Items[i]

		// Hard safety invariant: never touch anything that is not simulated.
		if !safety.IsSimulatedNode(node) {
			logger.V(1).Info("skipping non-simulated node", "node", node.Name)
			continue
		}
		matched++

		if pool.Spec.Advertise.DRA {
			if err := r.reconcileSlice(ctx, &pool, &model, node.Name); err != nil {
				return ctrl.Result{}, err
			}
			devices += pool.Spec.GPUsPerNode
		}
		if err := r.reconcileNode(ctx, &pool, &model, node); err != nil {
			return ctrl.Result{}, err
		}
	}

	pool.Status.ObservedGeneration = pool.Generation
	pool.Status.NodesMatched = matched
	pool.Status.DevicesPublished = devices
	r.setReady(&pool, metav1.ConditionTrue, "Reconciled",
		fmt.Sprintf("simulated %d GPUs across %d nodes", devices, matched))

	return ctrl.Result{}, r.Status().Update(ctx, &pool)
}

func (r *GPUPoolReconciler) reconcileSlice(ctx context.Context, pool *v1alpha1.GPUPool, model *v1alpha1.GPUModel, nodeName string) error {
	desired := gpu.BuildResourceSlice(pool, model, nodeName)

	existing := &resourcev1.ResourceSlice{ObjectMeta: metav1.ObjectMeta{Name: desired.Name}}
	_, err := controllerutil.CreateOrUpdate(ctx, r.Client, existing, func() error {
		existing.Spec = desired.Spec
		return controllerutil.SetControllerReference(pool, existing, r.Scheme)
	})
	return err
}

func (r *GPUPoolReconciler) reconcileNode(ctx context.Context, pool *v1alpha1.GPUPool, model *v1alpha1.GPUModel, node *corev1.Node) error {
	if pool.Spec.Advertise.ExtendedResource {
		qty := *resource.NewQuantity(int64(pool.Spec.GPUsPerNode), resource.DecimalSI)
		patched := node.DeepCopy()
		if patched.Status.Capacity == nil {
			patched.Status.Capacity = corev1.ResourceList{}
		}
		if patched.Status.Allocatable == nil {
			patched.Status.Allocatable = corev1.ResourceList{}
		}
		patched.Status.Capacity["nvidia.com/gpu"] = qty
		patched.Status.Allocatable["nvidia.com/gpu"] = qty
		if err := r.Status().Patch(ctx, patched, client.MergeFrom(node)); err != nil {
			return err
		}
		node = patched
	}

	labels := gpu.NodeLabels(pool, model)
	patched := node.DeepCopy()
	if patched.Labels == nil {
		patched.Labels = map[string]string{}
	}
	for k, v := range labels {
		patched.Labels[k] = v
	}
	return r.Patch(ctx, patched, client.MergeFrom(node))
}

func (r *GPUPoolReconciler) setReady(pool *v1alpha1.GPUPool, status metav1.ConditionStatus, reason, msg string) {
	meta.SetStatusCondition(&pool.Status.Conditions, metav1.Condition{
		Type:               "Ready",
		Status:             status,
		Reason:             reason,
		Message:            msg,
		ObservedGeneration: pool.Generation,
	})
}

// SetupWithManager wires the controller. Node changes re-trigger every pool,
// because a newly-created kwok node may match any pool's selector.
func (r *GPUPoolReconciler) SetupWithManager(mgr ctrl.Manager) error {
	return ctrl.NewControllerManagedBy(mgr).
		For(&v1alpha1.GPUPool{}).
		Owns(&resourcev1.ResourceSlice{}).
		Watches(&corev1.Node{}, handler.EnqueueRequestsFromMapFunc(r.poolsForNode)).
		Complete(r)
}

	if err := r.List(ctx, &pools); err != nil {
		return nil
	}
	reqs := make([]ctrl.Request, 0, len(pools.Items))
	for _, p := range pools.Items {
		reqs = append(reqs, ctrl.Request{NamespacedName: client.ObjectKey{Name: p.Name}})
	}
	return reqs
}

			podPhase(t, "e2e-gpu-b") == "Running"

// Package cli implements the ghostgpu command-line interface.
package cli

import (
	"fmt"

	"k8s.io/apimachinery/pkg/api/resource"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"sigs.k8s.io/controller-runtime/pkg/client"

	"github.com/santimillang/ghostgpu/api/v1alpha1"
)

// UpOptions describes a one-command simulated GPU fleet.
type UpOptions struct {
	Name        string
	Product     string
	Memory      string
	Compute     string
	GPUsPerNode int32
	NVLinkSize  int32
}

// BuildManifests turns CLI options into the GPUModel/GPUPool pair to apply.
func BuildManifests(opts UpOptions) ([]client.Object, error) {
	mem, err := resource.ParseQuantity(opts.Memory)
	if err != nil {
		return nil, fmt.Errorf("invalid memory %q: %w", opts.Memory, err)
	}

	model := &v1alpha1.GPUModel{
		ObjectMeta: metav1.ObjectMeta{Name: opts.Name},
		Spec: v1alpha1.GPUModelSpec{
			Vendor:            "nvidia",
			ProductName:       opts.Product,
			Memory:            mem,
			ComputeCapability: opts.Compute,
		},
	}

	pool := &v1alpha1.GPUPool{
		ObjectMeta: metav1.ObjectMeta{Name: opts.Name + "-pool"},
		Spec: v1alpha1.GPUPoolSpec{
			ModelRef:     opts.Name,
			NodeSelector: map[string]string{"type": "kwok"},
			GPUsPerNode:  opts.GPUsPerNode,
			Advertise:    v1alpha1.AdvertiseSpec{DRA: true, ExtendedResource: true},
			Topology:     v1alpha1.TopologySpec{NVLinkDomainSize: opts.NVLinkSize},
		},
	}

	return []client.Object{model, pool}, nil
}
```

// Command ghostgpu is the ghostgpu command-line interface.
package main

import (
	"context"
	"flag"
	"fmt"
	"os"

	ctrl "sigs.k8s.io/controller-runtime"
	"sigs.k8s.io/controller-runtime/pkg/client"

	"github.com/santimillang/ghostgpu/api/v1alpha1"
	"github.com/santimillang/ghostgpu/internal/cli"
)

func main() {
	if len(os.Args) < 2 || os.Args[1] != "up" {
		fmt.Fprintln(os.Stderr, "usage: ghostgpu up [flags]")
		os.Exit(2)
	}

	fs := flag.NewFlagSet("up", flag.ExitOnError)
	opts := cli.UpOptions{}
	fs.StringVar(&opts.Name, "name", "h100", "name for the GPUModel and GPUPool")
	fs.StringVar(&opts.Product, "gpu", "NVIDIA-H100-SXM", "GPU product name")
	fs.StringVar(&opts.Memory, "memory", "80Gi", "memory per GPU")
	fs.StringVar(&opts.Compute, "compute-capability", "9.0", "compute capability")
	var gpus, nvlink int
	fs.IntVar(&gpus, "gpus-per-node", 8, "simulated GPUs per node")
	fs.IntVar(&nvlink, "nvlink-domain-size", 4, "GPUs per NVLink domain (0 disables)")
	_ = fs.Parse(os.Args[2:])
	opts.GPUsPerNode = int32(gpus)
	opts.NVLinkSize = int32(nvlink)

	objs, err := cli.BuildManifests(opts)
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(1)
	}

	cfg, err := ctrl.GetConfig()
	if err != nil {
		fmt.Fprintln(os.Stderr, "error: no kubeconfig:", err)
		os.Exit(1)
	}
	c, err := client.New(cfg, client.Options{})
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(1)
	}
	_ = v1alpha1.AddToScheme(c.Scheme())

	for _, o := range objs {
		if err := c.Create(context.Background(), o); err != nil {
			fmt.Fprintln(os.Stderr, "error applying", o.GetName(), ":", err)
			os.Exit(1)
		}
		fmt.Println("created", o.GetObjectKind().GroupVersionKind().Kind, o.GetName())
	}
}
```
